Un conteo no es un indicador

Las cifras de salida son valores de referencia del curso, obtenidos con consultas reales de solo lectura sobre tablas equivalentes. Tus resultados deben coincidir si construiste Silver igual; los menús exactos de BigQuery Studio pueden verse distintos y no se ensayaron con una cuenta de alumno. Los bytes son los del estimador.

Antes de empezar

  • Un proyecto de Google Cloud con facturación activa y BigQuery Studio abierto.
  • Las tres tablas Silver de la sesión anterior en tu proyecto: tweets_silver.tweets (6,798,932 filas), tweets_silver.autores y tweets_silver.hashtags.
  • Permiso de editor de datos en tu proyecto (puedes crear datasets y tablas).
  • Cada paso lee solo unas decenas o cientos de MB de Silver; solo tema_dia llega a ≈ 1.4 GB.

BigQuery Studio → selecciona tu proyecto → Consulta SQL nueva → pega el bloque del paso → mira el estimador de bytes (arriba a la derecha del editor) → Ejecutar → compara el resultado con la salida esperada. Todas las tablas se crean con CREATE OR REPLACE TABLE: puedes repetir un paso sin duplicar filas.

Paso 1 · Temas y tabla de hechos del día · 15 min

Dónde: BigQuery Studio → Consulta SQL nueva. Elige tu proyecto en el selector superior. Abre el estimador de bytes (arriba a la derecha del editor).

Acción: Crea el dataset tweets_gold en la misma ubicación que tweets_silver, ejecuta 30_gold_tweets.sql completo (primero temas_palabras, luego tema_dia) y corre los tres controles. Lee el estimador antes de cada ejecución: tema_dia lee solo la columna texto de Silver. Para practicar: resuelve E1.

-- 2. Gold: un tweet cuenta una vez por tema, aunque use varias palabras del tema.
--    Grano: fecha + tema + grupo de idioma. pct_del_dia usa el total de tweets del mismo día e idioma.
CREATE OR REPLACE TABLE tweets_gold.tema_dia
PARTITION BY fecha AS
WITH por_tweet AS (
  SELECT t.fecha, p.tema,
         IF(t.idioma IN ('en', 'es'), t.idioma, 'otro') AS grupo_idioma,
         t.tweet_id, t.likes
  FROM tweets_silver.tweets AS t
  CROSS JOIN tweets_gold.temas_palabras AS p
  WHERE REGEXP_CONTAINS(LOWER(t.texto), CONCAT(r'\b', p.palabra, r'\b'))
  GROUP BY t.fecha, p.tema, grupo_idioma, t.tweet_id, t.likes
),
dia AS (
  SELECT fecha, IF(idioma IN ('en', 'es'), idioma, 'otro') AS grupo_idioma, COUNT(*) AS tweets_dia
  FROM tweets_silver.tweets
  GROUP BY fecha, grupo_idioma
)
SELECT m.fecha, m.tema, m.grupo_idioma,
       COUNT(*) AS tweets_tema, SUM(m.likes) AS likes_tema,
       ANY_VALUE(d.tweets_dia) AS tweets_dia,
       ROUND(100 * COUNT(*) / ANY_VALUE(d.tweets_dia), 3) AS pct_del_dia
FROM por_tweet AS m
JOIN dia AS d USING (fecha, grupo_idioma)
GROUP BY m.fecha, m.tema, m.grupo_idioma;

-- Requiere tweets_silver.tweets (sesión anterior). Lee solo la columna `texto` de Silver (≈ 1.2 GB).

WHERE tema = 'energia' AND grupo_idioma = 'en'

-- 1.3 Control del denominador: tweets_dia, sumado una vez por día e idioma, debe dar el total de Silver.

-- Gold · hashtags por día. Umbral: solo hashtags con al menos 25 tweets ese día (definición editable).
CREATE OR REPLACE TABLE tweets_gold.hashtags_dia

  SELECT fecha, COUNT(*) AS tweets_dia FROM tweets_silver.tweets GROUP BY fecha

JOIN por_dia AS d USING (fecha)
GROUP BY h.fecha, h.hashtag, d.tweets_dia

Paso 3 · Alcance de los autores · 10 min

Dónde: BigQuery Studio → Consulta SQL nueva.

-- Gold · concentración por tamaño de audiencia del autor (segmentos por seguidores).
CREATE OR REPLACE TABLE tweets_gold.autores_alcance AS
WITH base AS (
  SELECT t.tweet_id, t.likes, a.usuario,
         CASE WHEN a.seguidores < 1000      THEN '1 · menos de 1 mil'
              WHEN a.seguidores < 10000     THEN '2 · 1 mil a 10 mil'
              WHEN a.seguidores < 100000    THEN '3 · 10 mil a 100 mil'
              WHEN a.seguidores < 1000000   THEN '4 · 100 mil a 1 millón'
              ELSE                               '5 · más de 1 millón' END AS segmento
  FROM tweets_silver.tweets AS t
  JOIN tweets_silver.autores AS a USING (usuario)
)
SELECT segmento,
       COUNT(DISTINCT usuario) AS autores,
       COUNT(*)                AS tweets,
       SUM(likes)              AS likes,
       ROUND(100 * COUNT(*) / SUM(COUNT(*)) OVER (), 2) AS pct_tweets,
       ROUND(100 * SUM(likes) / SUM(SUM(likes)) OVER (), 2) AS pct_likes
FROM base
GROUP BY segmento;

-- PASO 3 · Alcance de los autores. Crea la tabla con 32_gold_autores_alcance.sql (une Silver tweets y autores).

-- 3.1 La tabla completa: cada fila es un segmento por seguidores del autor.
SELECT segmento, autores, tweets, likes, pct_tweets, pct_likes
FROM tweets_gold.autores_alcance
ORDER BY segmento;

-- 3.2 ¿Quiénes escriben mucho? Distribución de tweets por autor (tabla Silver autores).

  COUNT(*)                                       AS autores
FROM tweets_silver.autores;

-- Gold · desde qué aplicación se publica (top 8 del archivo; el resto se agrupa en «otras»).
CREATE OR REPLACE TABLE tweets_gold.aplicaciones_dia
PARTITION BY fecha AS
WITH top AS (
  SELECT aplicacion FROM tweets_silver.tweets GROUP BY aplicacion ORDER BY COUNT(*) DESC LIMIT 8
)
SELECT t.fecha,
       IF(t.aplicacion IN (SELECT aplicacion FROM top), t.aplicacion, 'otras') AS aplicacion,
       COUNT(*) AS tweets
FROM tweets_silver.tweets AS t
GROUP BY t.fecha, aplicacion;

-- Gold · ritmo del día: tweets por hora UTC y grupo de idioma.
CREATE OR REPLACE TABLE tweets_gold.actividad_hora
PARTITION BY fecha AS
SELECT fecha,
       EXTRACT(HOUR FROM fecha_hora) AS hora_utc,
       IF(idioma IN ('en', 'es'), idioma, 'otro') AS grupo_idioma,
       COUNT(*) AS tweets
FROM tweets_silver.tweets
GROUP BY fecha, hora_utc, grupo_idioma;

-- Gold · estructura de la conversación: originales, respuestas y citas por día, con likes típicos.
CREATE OR REPLACE TABLE tweets_gold.conversacion_dia
PARTITION BY fecha AS
SELECT fecha,
       CASE WHEN es_respuesta THEN 'respuesta'
            WHEN es_cita      THEN 'cita'
            ELSE                   'original' END AS tipo,
       COUNT(*) AS tweets,
       SUM(likes) AS likes,
       APPROX_QUANTILES(likes, 100)[OFFSET(50)] AS mediana_likes
FROM tweets_silver.tweets
GROUP BY fecha, tipo;

-- PASO 5 · Conversación y cuadre entre tablas. Crea la tabla con 35_gold_conversacion.sql.

-- 5.1 Originales, respuestas y citas en todo el periodo.
SELECT tipo, SUM(tweets) AS tweets, SUM(likes) AS likes,
       ROUND(SUM(likes) / SUM(tweets), 2) AS likes_por_tweet
FROM tweets_gold.conversacion_dia

ORDER BY tweets DESC;

Paso 6 · Precisión de las palabras clave y ficha de KPI · 13 min

-- PASO 6 · ¿Las palabras clave miden el tema? Lee 5 tweets por tema (23 de marzo, inglés y español).
-- Lee solo un día de texto de Silver (≈ 0.1 GB). Clasifica cada fila tú mismo: acierto, dudoso o falso positivo.
WITH m AS (
  SELECT p.tema, p.palabra, t.idioma,
         REPLACE(SUBSTR(t.texto, 1, 150), '\n', ' ') AS texto,
         ROW_NUMBER() OVER (PARTITION BY p.tema ORDER BY FARM_FINGERPRINT(CAST(t.tweet_id AS STRING))) AS fila
  FROM tweets_silver.tweets AS t
  CROSS JOIN tweets_gold.temas_palabras AS p
  WHERE t.fecha = '2022-03-23' AND t.idioma IN ('en', 'es')
    AND REGEXP_CONTAINS(LOWER(t.texto), CONCAT(r'\b', p.palabra, r'\b'))
)
SELECT tema, palabra, idioma, texto
FROM m
WHERE fila <= 5
ORDER BY tema, fila;

-- 2. Gold: un tweet cuenta una vez por tema, aunque use varias palabras del tema.
--    Grano: fecha + tema + grupo de idioma. pct_del_dia usa el total de tweets del mismo día e idioma.
CREATE OR REPLACE TABLE tweets_gold.tema_dia
PARTITION BY fecha AS
WITH por_tweet AS (
  SELECT t.fecha, p.tema,
         IF(t.idioma IN ('en', 'es'), t.idioma, 'otro') AS grupo_idioma,
         t.tweet_id, t.likes
  FROM tweets_silver.tweets AS t
  CROSS JOIN tweets_gold.temas_palabras AS p
  WHERE REGEXP_CONTAINS(LOWER(t.texto), CONCAT(r'\b', p.palabra, r'\b'))
  GROUP BY t.fecha, p.tema, grupo_idioma, t.tweet_id, t.likes
),
dia AS (
  SELECT fecha, IF(idioma IN ('en', 'es'), idioma, 'otro') AS grupo_idioma, COUNT(*) AS tweets_dia
  FROM tweets_silver.tweets
  GROUP BY fecha, grupo_idioma
)
SELECT m.fecha, m.tema, m.grupo_idioma,
       COUNT(*) AS tweets_tema, SUM(m.likes) AS likes_tema,
       ANY_VALUE(d.tweets_dia) AS tweets_dia,
       ROUND(100 * COUNT(*) / ANY_VALUE(d.tweets_dia), 3) AS pct_del_dia
FROM por_tweet AS m
JOIN dia AS d USING (fecha, grupo_idioma)
GROUP BY m.fecha, m.tema, m.grupo_idioma;

tweets/32_gold_autores_alcance.sql · ver contenido completo
-- Gold · concentración por tamaño de audiencia del autor (segmentos por seguidores).
CREATE OR REPLACE TABLE tweets_gold.autores_alcance AS
WITH base AS (
  SELECT t.tweet_id, t.likes, a.usuario,
         CASE WHEN a.seguidores < 1000      THEN '1 · menos de 1 mil'
              WHEN a.seguidores < 10000     THEN '2 · 1 mil a 10 mil'
              WHEN a.seguidores < 100000    THEN '3 · 10 mil a 100 mil'
              WHEN a.seguidores < 1000000   THEN '4 · 100 mil a 1 millón'
              ELSE                               '5 · más de 1 millón' END AS segmento
  FROM tweets_silver.tweets AS t
  JOIN tweets_silver.autores AS a USING (usuario)
)
SELECT segmento,
       COUNT(DISTINCT usuario) AS autores,
       COUNT(*)                AS tweets,
       SUM(likes)              AS likes,
       ROUND(100 * COUNT(*) / SUM(COUNT(*)) OVER (), 2) AS pct_tweets,
       ROUND(100 * SUM(likes) / SUM(SUM(likes)) OVER (), 2) AS pct_likes
FROM base
GROUP BY segmento;

tweets/s05_06_precision.sql · ver contenido completo
-- PASO 6 · ¿Las palabras clave miden el tema? Lee 5 tweets por tema (23 de marzo, inglés y español).
-- Lee solo un día de texto de Silver (≈ 0.1 GB). Clasifica cada fila tú mismo: acierto, dudoso o falso positivo.
WITH m AS (
  SELECT p.tema, p.palabra, t.idioma,
         REPLACE(SUBSTR(t.texto, 1, 150), '\n', ' ') AS texto,
         ROW_NUMBER() OVER (PARTITION BY p.tema ORDER BY FARM_FINGERPRINT(CAST(t.tweet_id AS STRING))) AS fila
  FROM tweets_silver.tweets AS t
  CROSS JOIN tweets_gold.temas_palabras AS p
  WHERE t.fecha = '2022-03-23' AND t.idioma IN ('en', 'es')
    AND REGEXP_CONTAINS(LOWER(t.texto), CONCAT(r'\b', p.palabra, r'\b'))
)
SELECT tema, palabra, idioma, texto
FROM m
WHERE fila <= 5
ORDER BY tema, fila;
