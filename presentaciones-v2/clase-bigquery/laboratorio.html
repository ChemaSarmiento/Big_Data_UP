01_ingestar_bronze.sql ↓

Consulta completa
-- Ejecutar tras cargar la muestra o el CSV completo en entrada.
-- Bronze conserva valores y duplicados del lote; entrada es el área de carga reemplazable.
-- Ejecución serial en proyecto dedicado: no protege contra dos escritores concurrentes.
DECLARE lote STRING DEFAULT 'all_data_muestra_v1';
DECLARE fuente STRING DEFAULT 'drive:1dLxKFxawXTTaqCDr9Y9YpYdWTpOhol42; muestra 4500 filas';
CREATE TABLE IF NOT EXISTS emergente_bronze.observaciones AS
SELECT e.*, CAST(NULL AS STRING) AS lote_id,
 CAST(NULL AS STRING) AS fuente_uri, CURRENT_TIMESTAMP() AS ingestado_en,
 TO_HEX(SHA256(TO_JSON_STRING(e))) AS fila_hash
FROM emergente_bronze.entrada e WHERE FALSE;
ASSERT (SELECT COUNT(*) > 0 FROM emergente_bronze.entrada) AS 'entrada está vacía';
ASSERT (SELECT COUNT(*) = 0 FROM emergente_bronze.observaciones WHERE lote_id=lote)
 AS 'Lote ya cargado: no lo repitas. Para otro archivo usa otro ID y su fuente.';
INSERT INTO emergente_bronze.observaciones
SELECT e.*, lote, fuente, CURRENT_TIMESTAMP(), TO_HEX(SHA256(TO_JSON_STRING(e)))
FROM emergente_bronze.entrada e;
SELECT lote_id, COUNT(*) AS filas, COUNT(DISTINCT fila_hash) AS filas_distintas
FROM emergente_bronze.observaciones GROUP BY lote_id;

03_silver.sql ↓

Consulta completa
-- Reconstrucción completa, reproducible: primera ruta docente, no actualización incremental.
-- Con el archivo entero, ensayar costo y duración antes de clase.
CREATE TEMP FUNCTION limpia(s STRING) AS (NULLIF(REGEXP_REPLACE(UPPER(TRIM(s)), r'\s+', ' '), ''));
CREATE TEMP TABLE unicas AS
SELECT * FROM emergente_bronze.observaciones
QUALIFY ROW_NUMBER() OVER(PARTITION BY fila_hash ORDER BY ingestado_en,lote_id)=1;
CREATE TEMP TABLE tipadas AS
SELECT fila_hash,lote_id,fuente_uri, producto AS producto_original,
 limpia(producto) AS producto, limpia(presentacion) AS presentacion,
 limpia(marca) AS marca, limpia(categoria) AS categoria, limpia(catalogo) AS catalogo,
 SAFE_CAST(precio AS NUMERIC) AS precio_mxn,
 SAFE_CAST(fechaRegistro AS DATETIME) AS fecha_hora_local,
 DATE(SAFE_CAST(fechaRegistro AS DATETIME)) AS fecha,
 limpia(cadenaComercial) AS cadena, limpia(nombreComercial) AS establecimiento,
 limpia(giro) AS giro, limpia(direccion) AS direccion,
 limpia(estado) AS estado, limpia(municipio) AS municipio,
 SAFE_CAST(latitud AS FLOAT64) AS latitud, SAFE_CAST(longitud AS FLOAT64) AS longitud
FROM unicas;
CREATE TEMP TABLE clasificadas AS
SELECT *, CASE
 WHEN fecha IS NULL THEN 'fecha_invalida'
 WHEN precio_mxn IS NULL OR precio_mxn<=0 THEN 'precio_invalido'
 WHEN producto IS NULL OR marca IS NULL OR presentacion IS NULL THEN 'producto_incompleto'
 WHEN establecimiento IS NULL OR direccion IS NULL OR estado IS NULL OR municipio IS NULL THEN 'tienda_incompleta'
 ELSE NULL END AS motivo
FROM tipadas;
CREATE OR REPLACE TABLE emergente_silver.rechazadas AS
SELECT * FROM clasificadas WHERE motivo IS NOT NULL;
CREATE OR REPLACE TABLE emergente_silver.precios
PARTITION BY DATE_TRUNC(fecha, MONTH)
CLUSTER BY estado,municipio,producto,marca
OPTIONS(require_partition_filter=true) AS
SELECT * EXCEPT(motivo),
 TO_HEX(SHA256(TO_JSON_STRING(STRUCT(producto,marca,presentacion)))) AS producto_id,
 TO_HEX(SHA256(TO_JSON_STRING(STRUCT(cadena,establecimiento,direccion,estado,municipio)))) AS tienda_id
FROM clasificadas WHERE motivo IS NULL;
-- Línea base de rendimiento SOLO para taller: misma Silver, sin partición ni clustering.
-- Duplica almacenamiento temporalmente. Expira en7 días; retirar al terminar comparación.
CREATE OR REPLACE TABLE emergente_silver.precios_plana AS
SELECT * EXCEPT(motivo),
 TO_HEX(SHA256(TO_JSON_STRING(STRUCT(producto,marca,presentacion)))) AS producto_id,
 TO_HEX(SHA256(TO_JSON_STRING(STRUCT(cadena,establecimiento,direccion,estado,municipio)))) AS tienda_id
FROM clasificadas WHERE motivo IS NULL;
CREATE OR REPLACE TABLE emergente_silver.control AS
SELECT (SELECT COUNT(*) FROM emergente_bronze.observaciones) AS entrada,
 (SELECT COUNT(*) FROM unicas) AS distintas,
 (SELECT COUNT(*) FROM emergente_bronze.observaciones)-(SELECT COUNT(*) FROM unicas) AS duplicadas_exactas,
 COUNTIF(motivo IS NULL) AS aceptadas,COUNTIF(motivo IS NOT NULL) AS rechazadas
FROM clasificadas;
ASSERT (SELECT entrada=duplicadas_exactas+aceptadas+rechazadas FROM emergente_silver.control)
 AS 'La reconciliación no cierra';
SELECT * FROM emergente_silver.control;
-- Los IDs son claves derivadas, no códigos oficiales de producto/tienda.
-- No fusionar automáticamente DISTRITO FEDERAL/CDMX ni presentaciones diferentes.
-- La fecha es local sin zona informada; no asumir UTC.

04_gold.sql ↓

Consulta completa
-- Elegir periodo compatible. Estos parámetros cubren la muestra, no certifican el archivo completo.
DECLARE inicio DATE DEFAULT DATE '2011-01-01';
DECLARE fin DATE DEFAULT DATE '2015-12-31';
CREATE OR REPLACE TABLE emergente_gold.precio_tienda_dia AS
SELECT fecha,estado,municipio,producto_id,producto,marca,presentacion,
 tienda_id,establecimiento,direccion,COUNT(*) AS observaciones,
 COUNT(DISTINCT precio_mxn) AS precios_distintos,
 MIN(precio_mxn) AS precio_min_mxn,MAX(precio_mxn) AS precio_max_mxn,
 IF(COUNT(DISTINCT precio_mxn)=1,MIN(precio_mxn),NULL) AS precio_comparable_mxn
FROM emergente_silver.precios
WHERE fecha BETWEEN inicio AND fin
GROUP BY fecha,estado,municipio,producto_id,producto,marca,presentacion,tienda_id,establecimiento,direccion;
CREATE OR REPLACE TABLE emergente_gold.comparacion AS
SELECT fecha,estado,municipio,producto_id,producto,marca,presentacion,
 COUNT(*) AS tiendas_observadas,
 COUNTIF(precio_comparable_mxn IS NOT NULL) AS tiendas_comparables,
 COUNTIF(precio_comparable_mxn IS NULL) AS tiendas_con_conflicto,
 MIN(precio_comparable_mxn) AS menor_precio_mxn,
 MAX(precio_comparable_mxn) AS mayor_precio_mxn,
 MAX(precio_comparable_mxn)-MIN(precio_comparable_mxn) AS diferencia_mxn
FROM emergente_gold.precio_tienda_dia
GROUP BY fecha,estado,municipio,producto_id,producto,marca,presentacion;
SELECT * FROM emergente_gold.comparacion
WHERE fecha=DATE '2014-03-19' AND estado='DISTRITO FEDERAL' AND municipio='TLALPAN'
 AND producto='REFRESCO' AND marca='SPRITE' AND presentacion='BOTELLA 2 LT. NO RETORNABLE';
-- Muestra:2 tiendas comparables,15.10 y15.50 MXN; diferencia0.40 MXN.
-- Esto no mide ahorro actual ni disponibilidad. La muestra no cubre todas las tiendas.
-- S/M es marca no especificada: no basta para afirmar equivalencia comercial.
