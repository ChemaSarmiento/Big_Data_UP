-- Paso 1.a · crea el dataset (ejecútalo primero).
CREATE SCHEMA IF NOT EXISTS tweets_silver OPTIONS (location = 'US');

-- Paso 1.b · crea la tabla Silver de tweets (20_silver_tweets.sql). Lee el estimador antes de ejecutar.
-- Silver: una fila por tweet, con tipos reales. Bronze no se modifica.
-- Cuesta un escaneo completo de Bronze (~23.7 GB facturables); ejecútalo una vez.
CREATE OR REPLACE TABLE tweets_silver.tweets
PARTITION BY fecha
CLUSTER BY idioma AS
SELECT
  SAFE_CAST(JSON_VALUE(value, '$.id') AS INT64)                  AS tweet_id,
  SAFE_CAST(JSON_VALUE(value, '$.date') AS TIMESTAMP)            AS fecha_hora,
  DATE(SAFE_CAST(JSON_VALUE(value, '$.date') AS TIMESTAMP))      AS fecha,
  JSON_VALUE(value, '$.lang')                                    AS idioma,
  JSON_VALUE(value, '$.content')                                 AS texto,
  JSON_VALUE(value, '$.url')                                     AS url,
  JSON_VALUE(value, '$.user.username')                           AS usuario,
  SAFE_CAST(JSON_VALUE(value, '$.user.verified') AS BOOL)        AS usuario_verificado,
  SAFE_CAST(JSON_VALUE(value, '$.user.followersCount') AS INT64) AS seguidores,
  JSON_VALUE(value, '$.user.location')                           AS ubicacion_usuario,
  SAFE_CAST(JSON_VALUE(value, '$.likeCount') AS INT64)           AS likes,
  SAFE_CAST(JSON_VALUE(value, '$.retweetCount') AS INT64)        AS retuits,
  SAFE_CAST(JSON_VALUE(value, '$.replyCount') AS INT64)          AS respuestas,
  SAFE_CAST(JSON_VALUE(value, '$.quoteCount') AS INT64)          AS citas,
  JSON_VALUE(value, '$.inReplyToTweetId') IS NOT NULL            AS es_respuesta,
  JSON_VALUE(value, '$.quotedTweet.id') IS NOT NULL              AS es_cita,
  ARRAY(SELECT LOWER(h) FROM UNNEST(JSON_VALUE_ARRAY(value, '$.hashtags')) AS h) AS hashtags,
  JSON_VALUE(value, '$.sourceLabel')                             AS aplicacion
FROM tweets_bronze.entrada;

Paso 3 · Crea autores: una fila por autor · 12 min

Dónde: BigQuery Studio → pestaña nueva.

Acción: Ejecuta 21_silver_autores.sql: usa un CTE (WITH) y ROW_NUMBER para quedarse con el dato más reciente de cada autor. Después ejecuta las cuatro consultas de s04_03_autores_checks.sql. En 3.1 mira cómo cambian los seguidores de skift y cuál fila recibe orden = 1.

Costo estimado: Referencia: 279,906,967 bytes (≈280 MB) para crear la tabla; las comprobaciones, ≈253 + 75 + 40 + 90 MB.

-- Silver 2 · autores: una fila por autor (grano distinto al de tweets).
-- Un autor aparece en muchos tweets; nos quedamos con su dato más reciente.
CREATE OR REPLACE TABLE tweets_silver.autores AS
WITH ordenados AS (
  SELECT usuario, usuario_verificado, seguidores,
         NULLIF(TRIM(ubicacion_usuario), '') AS ubicacion,
         ROW_NUMBER() OVER (PARTITION BY usuario ORDER BY fecha_hora DESC) AS orden,
         COUNT(*)     OVER (PARTITION BY usuario)                          AS tweets_en_archivo
  FROM tweets_silver.tweets
)
SELECT usuario, usuario_verificado, seguidores, ubicacion, tweets_en_archivo
FROM ordenados
WHERE orden = 1;

-- Paso 3 · Revisa la tabla autores (la creaste con 21_silver_autores.sql).

-- 3.1 Un autor con varios tweets, del más reciente al más antiguo.
--     orden = 1 es el que se queda en autores. Mira cómo cambia seguidores.
SELECT tweet_id, fecha_hora, seguidores,
       ROW_NUMBER() OVER (PARTITION BY usuario ORDER BY fecha_hora DESC) AS orden
FROM tweets_silver.tweets
WHERE usuario = 'skift'
ORDER BY fecha_hora DESC;

-- 3.2 Lo que quedó de ese autor: una sola fila.
SELECT * FROM tweets_silver.autores WHERE usuario = 'skift';

-- 3.3 ¿Cuadra? Una fila por autor y todos los tweets contados una vez.
SELECT COUNT(*)                  AS autores,
       COUNT(DISTINCT usuario)   AS usuarios_distintos,
       SUM(tweets_en_archivo)    AS tweets_sumados
FROM tweets_silver.autores;

-- 3.4 CTE: WITH le pone nombre a un paso intermedio para poder resumirlo otra vez.
--     Primero cuenta tweets por autor; después cuenta autores por tramo.
WITH tweets_por_autor AS (
  SELECT usuario, COUNT(*) AS tweets
  FROM tweets_silver.tweets
  GROUP BY usuario
)
SELECT CASE WHEN tweets = 1  THEN '1'
            WHEN tweets = 2  THEN '2'
            WHEN tweets <= 5 THEN '3 a 5'
            WHEN tweets <= 20 THEN '6 a 20'
            ELSE 'más de 20' END AS tweets_publicados,
       MIN(tweets)   AS orden,
       COUNT(*)      AS autores,
       SUM(tweets)   AS tweets_totales
FROM tweets_por_autor
GROUP BY tweets_publicados
ORDER BY orden;

-- Silver 3 · hashtags: una fila por tweet y hashtag. UNNEST convierte el arreglo en filas.
-- DISTINCT evita contar dos veces un hashtag repetido dentro del mismo tweet.
CREATE OR REPLACE TABLE tweets_silver.hashtags
PARTITION BY fecha AS
SELECT DISTINCT t.tweet_id, t.fecha, t.idioma, h AS hashtag
FROM tweets_silver.tweets AS t, UNNEST(t.hashtags) AS h
WHERE h != '';

Paso 5 · Marca la calidad sin borrar nada · 12 min

-- Paso 5 · Marca, no borres. Cada regla es una columna verdadero/falso; tú decides después qué excluir.

-- 5.1 Cuántos tweets cae en cada regla y cuántos quedan al aplicarlas.
--     texto_repetido: el mismo texto exacto aparece 20 veces o más.
WITH marcadas AS (
  SELECT fecha = DATE '2022-03-16'                  AS dia_parcial,
         idioma = 'und'                             AS idioma_desconocido,
         COUNT(*) OVER (PARTITION BY texto) >= 20   AS texto_repetido,
         es_respuesta
  FROM tweets_silver.tweets
)
SELECT COUNT(*)                                                                         AS tweets,
       COUNTIF(dia_parcial)                                                             AS dia_parcial,
       COUNTIF(idioma_desconocido)                                                      AS idioma_desconocido,
       COUNTIF(texto_repetido)                                                          AS texto_repetido,
       COUNTIF(NOT (dia_parcial OR idioma_desconocido OR texto_repetido))               AS quedan_con_3_reglas,
       COUNTIF(NOT (dia_parcial OR idioma_desconocido OR texto_repetido OR es_respuesta)) AS quedan_sin_respuestas
FROM marcadas;

-- 5.2 Los textos que más se copian. El nombre del paso (por_texto) es distinto al de la columna (texto):
--     si el alias de SELECT se llamara igual que la columna, GROUP BY agruparía por el alias recortado.
WITH por_texto AS (
  SELECT texto, COUNT(*) AS copias, COUNT(DISTINCT usuario) AS cuentas
  FROM tweets_silver.tweets
  GROUP BY texto
)
SELECT SUBSTR(texto, 1, 40) AS inicio, copias, cuentas
FROM por_texto
ORDER BY copias DESC
LIMIT 4;

-- 5.3 Ubicación vacía e idioma sin determinar.
SELECT COUNTIF(ubicacion_usuario IS NULL OR TRIM(ubicacion_usuario) = '')                        AS sin_ubicacion,
       ROUND(100 * COUNTIF(ubicacion_usuario IS NULL OR TRIM(ubicacion_usuario) = '') / COUNT(*), 1) AS pct_sin_ubicacion,
       COUNTIF(idioma = 'und')                                                                   AS idioma_und
FROM tweets_silver.tweets;

-- 5.4 Likes: el promedio y la mediana cuentan historias distintas.
SELECT ROUND(AVG(likes), 2)                          AS promedio,
       APPROX_QUANTILES(likes, 100)[OFFSET(50)]      AS mediana,
       ROUND(100 * COUNTIF(likes = 0) / COUNT(*), 1) AS pct_con_cero_likes,
       COUNTIF(likes >= 1000)                        AS tweets_con_mil_o_mas,
       ROUND(100 * SUM(IF(likes >= 1000, likes, 0)) / SUM(likes), 1) AS pct_de_los_likes
FROM tweets_silver.tweets;

-- Silver 2 · autores: una fila por autor (grano distinto al de tweets).
-- Un autor aparece en muchos tweets; nos quedamos con su dato más reciente.
CREATE OR REPLACE TABLE tweets_silver.autores AS
WITH ordenados AS (
  SELECT usuario, usuario_verificado, seguidores,
         NULLIF(TRIM(ubicacion_usuario), '') AS ubicacion,
         ROW_NUMBER() OVER (PARTITION BY usuario ORDER BY fecha_hora DESC) AS orden,
         COUNT(*)     OVER (PARTITION BY usuario)                          AS tweets_en_archivo
  FROM tweets_silver.tweets
)
SELECT usuario, usuario_verificado, seguidores, ubicacion, tweets_en_archivo
FROM ordenados
WHERE orden = 1;

-- Paso 3 · Revisa la tabla autores (la creaste con 21_silver_autores.sql).

-- 3.1 Un autor con varios tweets, del más reciente al más antiguo.
--     orden = 1 es el que se queda en autores. Mira cómo cambia seguidores.

       ROW_NUMBER() OVER (PARTITION BY usuario ORDER BY fecha_hora DESC) AS orden
FROM tweets_silver.tweets
WHERE usuario = 'skift'
ORDER BY fecha_hora DESC;

-- 3.2 Lo que quedó de ese autor: una sola fila.
SELECT * FROM tweets_silver.autores WHERE usuario = 'skift';

-- 3.3 ¿Cuadra? Una fila por autor y todos los tweets contados una vez.
SELECT COUNT(*)                  AS autores,
       COUNT(DISTINCT usuario)   AS usuarios_distintos,
       SUM(tweets_en_archivo)    AS tweets_sumados
FROM tweets_silver.autores;

-- 3.4 CTE: WITH le pone nombre a un paso intermedio para poder resumirlo otra vez.
--     Primero cuenta tweets por autor; después cuenta autores por tramo.
WITH tweets_por_autor AS (
  SELECT usuario, COUNT(*) AS tweets
  FROM tweets_silver.tweets
  GROUP BY usuario
)
SELECT CASE WHEN tweets = 1  THEN '1'
            WHEN tweets = 2  THEN '2'
            WHEN tweets <= 5 THEN '3 a 5'
            WHEN tweets <= 20 THEN '6 a 20'
            ELSE 'más de 20' END AS tweets_publicados,
       MIN(tweets)   AS orden,
       COUNT(*)      AS autores,
       SUM(tweets)   AS tweets_totales
FROM tweets_por_autor
GROUP BY tweets_publicados
ORDER BY orden;

tweets/s04_05_marcas_calidad.sql · ver contenido completo
-- Paso 5 · Marca, no borres. Cada regla es una columna verdadero/falso; tú decides después qué excluir.

-- 5.1 Cuántos tweets cae en cada regla y cuántos quedan al aplicarlas.
--     texto_repetido: el mismo texto exacto aparece 20 veces o más.
WITH marcadas AS (
  SELECT fecha = DATE '2022-03-16'                  AS dia_parcial,
         idioma = 'und'                             AS idioma_desconocido,
         COUNT(*) OVER (PARTITION BY texto) >= 20   AS texto_repetido,
         es_respuesta
  FROM tweets_silver.tweets
)
SELECT COUNT(*)                                                                         AS tweets,
       COUNTIF(dia_parcial)                                                             AS dia_parcial,
       COUNTIF(idioma_desconocido)                                                      AS idioma_desconocido,
       COUNTIF(texto_repetido)                                                          AS texto_repetido,
       COUNTIF(NOT (dia_parcial OR idioma_desconocido OR texto_repetido))               AS quedan_con_3_reglas,
       COUNTIF(NOT (dia_parcial OR idioma_desconocido OR texto_repetido OR es_respuesta)) AS quedan_sin_respuestas
FROM marcadas;

-- 5.2 Los textos que más se copian. El nombre del paso (por_texto) es distinto al de la columna (texto):
--     si el alias de SELECT se llamara igual que la columna, GROUP BY agruparía por el alias recortado.
WITH por_texto AS (
  SELECT texto, COUNT(*) AS copias, COUNT(DISTINCT usuario) AS cuentas
  FROM tweets_silver.tweets
  GROUP BY texto
)
SELECT SUBSTR(texto, 1, 40) AS inicio, copias, cuentas
FROM por_texto
ORDER BY copias DESC
LIMIT 4;

-- 5.3 Ubicación vacía e idioma sin determinar.
SELECT COUNTIF(ubicacion_usuario IS NULL OR TRIM(ubicacion_usuario) = '')                        AS sin_ubicacion,
       ROUND(100 * COUNTIF(ubicacion_usuario IS NULL OR TRIM(ubicacion_usuario) = '') / COUNT(*), 1) AS pct_sin_ubicacion,
       COUNTIF(idioma = 'und')                                                                   AS idioma_und
FROM tweets_silver.tweets;

-- 5.4 Likes: el promedio y la mediana cuentan historias distintas.
SELECT ROUND(AVG(likes), 2)                          AS promedio,
       APPROX_QUANTILES(likes, 100)[OFFSET(50)]      AS mediana,
       ROUND(100 * COUNTIF(likes = 0) / COUNT(*), 1) AS pct_con_cero_likes,
       COUNTIF(likes >= 1000)                        AS tweets_con_mil_o_mas,
       ROUND(100 * SUM(IF(likes >= 1000, likes, 0)) / SUM(likes), 1) AS pct_de_los_likes
FROM tweets_silver.tweets;
